### 5.2 MVP Recommendations Logic (Simplified)

```typescript
// Simplified recommendation engine for MVP
// Based on visibility score and basic checks

interface Recommendation {
  id: string;
  priority: 'high' | 'medium' | 'low';
  title: string;
  description: string;
  effort: string;
  timeline: string;
  expectedImpact: string;
  actionUrl?: string;
  category: 'entity' | 'content' | 'authority' | 'technical';
}

function generateRecommendations(
  visibilityScore: number,
  mentionData: MentionAnalysis,
  brandConfig: Brand
): Recommendation[] {
  const recommendations: Recommendation[] = [];

  // Rule-based recommendations based on score ranges
  if (visibilityScore < 30) {
    // Critical: Foundation issues
    recommendations.push({
      id: 'foundation-1',
      priority: 'high',
      title: 'Complete your Google Business Profile',
      description: 'Your AI visibility is very low. Start with Google Business Profile as it\'s the foundation for AI entity recognition.',
      effort: '1-2 hours',
      timeline: '1-2 weeks to see impact',
      expectedImpact: '+5-10% visibility',
      actionUrl: 'https://business.google.com/',
      category: 'entity'
    });
  }

  // Check for FAQ content
  if (!mentionData.hasFAQContent) {
    recommendations.push({
      id: 'content-1',
      priority: 'high',
      title: 'Add FAQ section to your website',
      description: 'AI systems cite FAQ content 30% more often. Add 6-10 common questions with direct answers.',
      effort: '4-6 hours',
      timeline: '1-2 weeks',
      expectedImpact: '+10-15% AI citation rate',
      category: 'content'
    });
  }

  // Check sentiment
  if (mentionData.averageSentiment < 0.3) {
    recommendations.push({
      id: 'authority-1',
      priority: 'medium',
      title: 'Improve brand sentiment through reviews',
      description: 'Your AI mentions have neutral/negative sentiment. Generate positive reviews on G2, Capterra, or Google.',
      effort: 'Ongoing',
      timeline: '4-8 weeks',
      expectedImpact: '+5-10% positive sentiment shift',
      category: 'authority'
    });
  }

  // Competitor-based recommendations
  if (mentionData.competitorMentionRate > mentionData.brandMentionRate) {
    recommendations.push({
      id: 'competitive-1',
      priority: 'high',
      title: 'Create comparison content',
      description: `Competitors are mentioned ${Math.round((mentionData.competitorMentionRate / mentionData.brandMentionRate - 1) * 100)}% more often. Create "Your Brand vs Competitor" comparison pages.`,
      effort: '8-10 hours per page',
      timeline: '3-4 weeks',
      expectedImpact: '+15-20% share of voice',
      category: 'content'
    });
  }

  // Always recommend: LinkedIn presence
  recommendations.push({
    id: 'entity-2',
    priority: 'medium',
    title: 'Optimize LinkedIn company page',
    description: 'LinkedIn is the #1 most-cited domain for B2B queries. Ensure complete About section, regular posts, and employee engagement.',
    effort: '2-3 hours + ongoing',
    timeline: '2-4 weeks',
    expectedImpact: '+5-10% B2B visibility',
    actionUrl: 'https://linkedin.com/company/',
    category: 'entity'
  });

  // Sort by priority
  return recommendations.sort((a, b) => {
    const priorityOrder = { high: 0, medium: 1, low: 2 };
    return priorityOrder[a.priority] - priorityOrder[b.priority];
  });
}
```
